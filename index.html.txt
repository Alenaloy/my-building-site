    <script src="https://cdn.jsdelivr.net/npm/three@0.128.0/build/three.min.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/controls/OrbitControls.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/loaders/GLTFLoader.js"></script>
    <script src="https://cdn.jsdelivr.net/npm/three@0.128.0/examples/js/loaders/DRACOLoader.js"></script>

    <script>
        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(45, window.innerWidth/window.innerHeight, 0.1, 5000);
        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        document.getElementById('canvas-container').appendChild(renderer.domElement);
        
        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        scene.add(new THREE.AmbientLight(0xffffff, 2));
        camera.position.set(0, 50, 150);

        // ВАЖНО: Настройка Draco для сжатых моделей
        const loader = new THREE.GLTFLoader();
        const dracoLoader = new THREE.DRACOLoader();
        // Используем официальный декодер от Google
        dracoLoader.setDecoderPath('https://www.gstatic.com/draco/versioned/decoders/1.4.1/');
        loader.setDRACOLoader(dracoLoader);
        
        // Загружаем именно scan3.glb
        loader.load('scan3.glb', (gltf) => {
            console.log("Успех! Модель scan3.glb загружена.");
            
            gltf.scene.traverse(obj => {
                if(obj.isMesh) {
                    // Создаем сетку поверх оригинальной модели
                    const wireframe = new THREE.Mesh(
                        obj.geometry,
                        new THREE.MeshBasicMaterial({
                            color: 0xffffff,
                            wireframe: true,
                            transparent: true,
                            opacity: 0.3
                        })
                    );
                    obj.add(wireframe);
                    obj.material.color.set(0x333333);
                }
            });
            scene.add(gltf.scene);
        }, undefined, (err) => {
            console.error("Ошибка загрузки:", err);
            alert("Ошибка: " + err.message);
        });

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }
        animate();
    </script>
